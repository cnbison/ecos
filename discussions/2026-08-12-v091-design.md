## 1. 背景与定位

### 1.1 为什么是 v0.91

v0.90.0-d 完成 Phase 7+ 抽象推演 #3 (POMDP T/R 在线学习 + Runtime + 冷启动). POMDP Policy 现在有:

- ✅ 4 状态 + Bayesian belief
- ✅ 依赖型 T(s'|s, a) (n_states × n_states × n_arms)
- ✅ R(s, a) Beta posterior (v0.90.0-a/b)
- ✅ bayes_update(action, observation)
- ✅ PBVI 完整算法 (α-vector + backup + 收敛)
- ✅ POMDPPolicy 默认 use_pbvi=True + use_learned_t_r=True + min_samples=5 冷启动
- ✅ Runtime + PolicyABTest 集成 + 3-way A/B

v0.90 final 留下的 4 个候选缺口 (per 12-kernel-mapping §8.2):

- ❌ Twin → Human Twin 抽象 (Cognitive Twin Agent 升级, 当前 Twin 仅 BeliefState + Trajectory)
- ⚠️ Plugin SDK 文档化 (v0.85.0-d 4 endpoint 仅 produce event, 无 Runtime subscriber)
- ❌ Teacher/Parent Dashboard (Kernel-first 主动推迟项, Bisen 2026-08-06 拍板)
- ⚠️ POMDP T/R 后验可视化 (display layer, 留 v0.91+)

v0.91 范围: Twin → Human Twin 抽象 + Plugin SDK 文档化. Teacher/Parent Dashboard 跟 Kernel-first 战略矛盾 (留 Kernel 稳定后). POMDP T/R 后验可视化留 v0.92+ (分量小不阻塞主轴).

### 2.3 HumanFeedbackEntry 数据结构

```python
@dataclass(frozen=True)
class HumanFeedbackEntry:
    """Human-in-loop 信号 (v0.91.0-a).

    4 event_type: hint_requested / idle_detected / goal_changed / reflection_completed.
    source 默认 "plugin" (Plugin SDK 4 endpoint), 留 v0.92+ 扩展 "teacher" / "parent".
    """
    student_id: str
    timestamp: datetime
    event_type: str  # 4 值枚举
    payload: Dict[str, Any]
    source: str = "plugin"
    schema_version: str = "0.91.0"

    def to_dict(self) -> Dict[str, Any]:
        return {
            "student_id": self.student_id,
            "timestamp": self.timestamp.isoformat(),
            "event_type": self.event_type,
            "payload": self.payload,
            "source": self.source,
            "schema_version": self.schema_version,
        }

    @classmethod
    def from_dict(cls, state: Dict[str, Any]) -> "HumanFeedbackEntry":
        if state.get("schema_version") != "0.91.0":
            raise ValueError(f"Unsupported schema_version: {state.get('schema_version')}")
        return cls(
            student_id=state["student_id"],
            timestamp=datetime.fromisoformat(state["timestamp"]),
            event_type=state["event_type"],
            payload=state["payload"],
            source=state.get("source", "plugin"),
        )
```

### 2.4 HumanFeedbackTrajectory + CognitiveTwinAgent 数据结构

```python
@dataclass
class HumanFeedbackTrajectory:
    """Human feedback 轨迹 (v0.91.0-a). 跟 TrajectoryState 同 pattern."""
    entries: List[HumanFeedbackEntry] = field(default_factory=list)
    maxlen: int = 500  # 跟 TrajectoryState 一致

    def append(self, entry: HumanFeedbackEntry) -> None:
        self.entries.append(entry)
        if len(self.entries) > self.maxlen:
            self.entries = self.entries[-self.maxlen:]

    def last_n(self, n: int) -> List[HumanFeedbackEntry]:
        return self.entries[-n:]

    def count_by_type(self, event_type: str) -> int:
        return sum(1 for e in self.entries if e.event_type == event_type)


@dataclass
class CognitiveTwinAgent:
    """Twin → Human Twin 3-tuple 聚合层 (v0.91.0-a).

    聚合 BeliefState + Trajectory + HumanFeedbackTrajectory, 跟 v0.83 Evidence Engine 同 pattern.
    ActionHistory 留 v0.92+ 占位.
    """
    belief_state: BeliefState
    trajectory: TrajectoryState
    human_feedback: HumanFeedbackTrajectory = field(default_factory=HumanFeedbackTrajectory)
    action_history: Optional[Dict[str, Any]] = None  # 占位 v0.92+
    schema_version: str = "0.91.0"

    @staticmethod
    def from_state(state: BeliefState) -> "CognitiveTwinAgent":
        return CognitiveTwinAgent(
            belief_state=state,
            trajectory=state.trajectory,
            human_feedback=HumanFeedbackTrajectory(),
        )

    def append_human_feedback(self, entry: HumanFeedbackEntry) -> None:
        """Allowlisted mutation site (defensive check [8])."""
        self.human_feedback.append(entry)
```

### 3.5 Runtime.plan_human_feedback_aware

```python
# ecos/runtime/api.py
def plan_human_feedback_aware(
    student_id: str,
    audience: str = "student",
    human_feedback_entry: Optional[HumanFeedbackEntry] = None,
    **kwargs,
) -> LCAResult:
    """v0.91.0-b: Human feedback 接入 Runtime.plan (6 plan API 上限).

    调 LCAEngine.append_human_feedback → LCAEngine.select_intervention(cognitive_twin=...).
    """
    lca = kwargs.get("lca_engine")
    cta_input = kwargs.get("cta_input")
    if human_feedback_entry is not None and lca is not None:
        lca.append_human_feedback(student_id, human_feedback_entry)
    cognitive_twin = lca._cognitive_twin.get(student_id) if lca is not None else None
    return lca.select_intervention(
        cta_input, audience=audience,
        motivation=kwargs.get("motivation"),
        domain_name=kwargs.get("domain_name"),
        cognitive_twin=cognitive_twin,
    )
```

### 4.3 _human_feedback_itype_override 算法

```python
@staticmethod
def _human_feedback_itype_override(cognitive_twin: Optional["CognitiveTwinAgent"]) -> Optional[InterventionType]:
    if cognitive_twin is None:
        return None
    counts = {
        "hint_requested": cognitive_twin.human_feedback.count_by_type("hint_requested"),
        "idle_detected": cognitive_twin.human_feedback.count_by_type("idle_detected"),
        "reflection_completed": cognitive_twin.human_feedback.count_by_type("reflection_completed"),
        "goal_changed": cognitive_twin.human_feedback.count_by_type("goal_changed"),
    }
    if counts["hint_requested"] > 5:
        return InterventionType.EXPLANATORY  # 主动求助 → 降难度
    if counts["idle_detected"] > 3:
        return InterventionType.INQUIRY      # 走神 → 激活兴趣
    if counts["reflection_completed"] > 3:
        return InterventionType.PRACTICE     # 主动反思 → 巩固
    if counts["goal_changed"] > 1:
        return InterventionType.PRACTICE     # 目标调整 → 巩固
    return None  # 走 default itype
```

```python
@dataclass
class CognitiveTwinAgent:
    # ... existing fields ...

    def dump_state(self) -> Dict[str, Any]:
        return {
            "state_dict": self.belief_state.to_dict(),
            "trajectory_to_dict": self.trajectory.to_dict() if hasattr(self.trajectory, "to_dict") else {},
            "human_feedback_to_dict": [e.to_dict() for e in self.human_feedback.entries],
            "action_history": self.action_history,
            "schema_version": "0.91.0",
        }

    @classmethod
    def load_state(cls, state: Dict[str, Any]) -> "CognitiveTwinAgent":
        schema_version = state.get("schema_version")
        if schema_version != "0.91.0":
            raise ValueError(f"Unsupported CognitiveTwinAgent schema_version: {schema_version!r}")
        belief_state = BeliefState.from_dict(state["state_dict"])
        trajectory = belief_state.trajectory  # already embedded
        entries = [HumanFeedbackEntry.from_dict(e) for e in state["human_feedback_to_dict"]]
        human_feedback = HumanFeedbackTrajectory(entries=entries)
        return cls(
            belief_state=belief_state,
            trajectory=trajectory,
            human_feedback=human_feedback,
            action_history=state.get("action_history"),
        )
```
